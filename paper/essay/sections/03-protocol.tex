\section{TCP Coordination Protocol and Message Types}\label{sec:protocol}
\ph{Main idea: master and workers coordinate over raw TCP with a line-oriented
protocol and pull-based work handout. Then, in order: each message is one JSON
object terminated by a newline, so framing is trivial and the traffic is
human-readable during debugging; the master never blocks a fast worker because
chunks are handed out on request, so faster nodes request more often;
\autoref{tab:msgs} lists the message types. Close on: newline-delimited JSON over
raw TCP was chosen for trivial framing rather than a length-prefixed or heavier
RPC scheme.}

\begin{table}[H]
\centering
\footnotesize
\begin{tabularx}{\linewidth}{l l X}
\hline
\textbf{Message} & \textbf{Direction} & \textbf{Purpose / fields}\\
\hline
\texttt{HELLO}     & W$\rightarrow$M & join; worker id, host, pid, cores\\
\texttt{PARAMS}    & M$\rightarrow$W & target data, digest, alphabet, length, construction\\
\texttt{REQUEST}   & W$\rightarrow$M & ask for a keyspace chunk\\
\texttt{ASSIGN}    & M$\rightarrow$W & \{chunk id, start, end\}\\
\texttt{PROGRESS}  & W$\rightarrow$M & \{chunk, tried, rate\}; doubles as a heartbeat\\
\texttt{HEARTBEAT} & W$\rightarrow$M & liveness while idle between chunks\\
\texttt{DONE}      & W$\rightarrow$M & chunk exhausted with no match\\
\texttt{FOUND}     & W$\rightarrow$M & \{chunk, key, index\} on a match\\
\texttt{NOWORK}    & M$\rightarrow$W & queue momentarily empty; retry\\
\texttt{STOP}      & M$\rightarrow$W & global halt (key found, or keyspace exhausted)\\
\hline
\end{tabularx}
\caption{Master--worker message protocol.}
\label{tab:msgs}
\end{table}

\paragraph{Lifecycle.} \ph{Main idea: the per-worker message lifecycle and how a
found key halts the cluster. Then, in order: a worker sends \texttt{HELLO},
receives \texttt{PARAMS}, then loops \texttt{REQUEST}$\rightarrow$\texttt{ASSIGN},
cracking each chunk in sub-batches while emitting \texttt{PROGRESS}, and reports
\texttt{DONE} or \texttt{FOUND}; on \texttt{FOUND} the master re-verifies the
claimed key against the captured digest before trusting it, then broadcasts
\texttt{STOP} so no node keeps hashing; the master is never idle, launching local
worker subprocesses that connect back over loopback and crack alongside the
remote workers. Close on: re-verification plus \texttt{STOP} guarantees a single
trusted result and an immediate global halt.}
